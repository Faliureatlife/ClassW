//1-7 are Primitive operations
//8 is max (first declared), holding the proc from the second line
//(9 is y=3) (second declared)
//10 is X=4
//11 is Y=6
//when Max X is ran at the end it adds the local values to the end of (8), 
//of [("GT",4),("Y",9)], showing that at that area X is greater than Y, (4>6)

local Max Y in
  Max = proc {$ X}
    local B in 
        {GT X Y B}
        if B then
          skip Browse X
        else
          skip Browse Y
        end
    end
  end
  Y = 3
  local X Y in 
    Y = 6
    X = 4
    skip Full
    {Max X}
  end
end
